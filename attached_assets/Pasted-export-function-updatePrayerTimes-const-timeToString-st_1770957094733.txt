export function updatePrayerTimes() {
    const timeToString = string => {
        return string.slice(0, -2); // get rid of am/pm
    }
    fetch("https://masjidal.com/api/v1/time/range?masjid_id=xwLVMDKJ", {"method": "get"})
	.then( (httpResponse) => {
		if (httpResponse.ok) {
		return httpResponse.json();
		} else {
		return Promise.reject("Fetch did not succeed");
		}
	} )
	.then(data => {
		console.log('data', data)
		const adhans = data['data']['salah'][0]
		const iqamas = data['data']['iqamah'][0]

		let fajrUpdate = {
			"_id": "79aa05ec-4920-4e71-a6e1-08e25139fa4e",
			"title": "Fajr",
			"adhan": timeToString(adhans['fajr']),
			"iqama": timeToString(iqamas['fajr']),
			"order": 1
		};
		let sunriseUpdate = {
			"_id": "5d07b78f-659d-457b-83c7-be9aeb5be097",
			"title": "Sunrise",
			"adhan": timeToString(adhans['sunrise']),
			"order": 2
		};
		let dhuhrUpdate = {
			"_id": "903950ae-fbbd-49d9-9c97-ae833b64db60",
			"title": "Dhuhr",
			"adhan": timeToString(adhans['zuhr']),
			"iqama": timeToString(iqamas['zuhr']),
			"order": 3
		};
		let asrUpdate = {
			"_id": "2a47ec17-027f-4749-86f2-fea6e1dc51ea",
			"title": "Asr",
			"adhan": timeToString(adhans['asr']),
			"iqama": timeToString(iqamas['asr']),
			"order": 4
		};
		let maghribUpdate = {
			"_id": "05506908-db80-40c8-9dd1-d0bc26fd03a2",
			"title": "Maghrib",
			"adhan": timeToString(adhans['maghrib']),
			"iqama": timeToString(iqamas['maghrib']),
			"order": 5
		};
		let ishaUpdate = {
			"_id": "04102469-dce5-4547-a06f-320d1880936a",
			"title": "Isha",
			"adhan": timeToString(adhans['isha']),
			"iqama": timeToString(iqamas['isha']),
			"order": 6
		};
		console.log('new prayer times')
		console.log(fajrUpdate)
		console.log(sunriseUpdate)
		console.log(dhuhrUpdate)
		console.log(asrUpdate)
		console.log(maghribUpdate)
		console.log(ishaUpdate)

		wixData.bulkUpdate("PrayerTimes", [fajrUpdate, sunriseUpdate, dhuhrUpdate, asrUpdate, maghribUpdate, ishaUpdate])
		.then((results) => {
			let updated = results.updated; // 2
			console.log('updated', updated)
		})
		.catch((err) => {
			console.log('error with bulk update', err)
		});
	})
	.catch(err => console.log(err));
}

updatePrayerTimes()
